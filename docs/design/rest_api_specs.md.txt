 ### 1. REST API Specification (Contract First)
  #### POST /api/v1/auth/tenants/{tenantCode}/register
  Registers a new gym client under the specified tenant scope (functional_and_non_functional.md:7,
  domain_model.md:127).
  #### Request Body (application/json):

    {
      "email": "client@example.com",
      "password": "SecurePassword123!",
      "fullName": "Juan Perez",
      "phoneNumber": "+5491122334455",
      "emergencyContact": "Maria Perez (+5491199887766)"
    }

  #### Response Responses:
  • 201 Created (Success):
    {
      "clientId": "a1b2c3d4-e5f6-7a8b-9c0d-1e2f3a4b5c6d",
      "tenantCode": "GYM_CENTRAL",
      "email": "client@example.com",
      "fullName": "Juan Perez",
      "accountStatus": "PENDING_PAYMENT",
      "accessToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
      "tokenType": "Bearer",
      "nextStep": "SELECT_MEMBERSHIP_PLAN"
    }
  • 400 Bad Request: Validation failure (e.g., weak password, invalid email format).
  • 404 Not Found: Gym tenant code does not exist.
  • 409 Conflict: Email is already registered within this tenant.
  • 422 Unprocessable Entity: Gym tenant is inactive.

    sequenceDiagram
        autonumber
        actor Mobile as Mobile App Client
        participant Controller as ClientAuthController
        participant Service as ClientRegistrationService

        Mobile->>Controller: POST /api/v1/auth/tenants/{tenantCode}/register (JSON)
        Controller->>Service: execute(RegisterClientCommand)

        Service->>TenantRepo: findByTenantCode(tenantCode)
        TenantRepo->>DB: SELECT * FROM gym_tenants WHERE tenant_code = ?
        DB-->>TenantRepo: GymTenant record
        TenantRepo-->>Service: GymTenant instance

        alt Tenant Not Found or Inactive
            Service-->>Controller: throw TenantNotFoundException / TenantInactiveException
            Controller-->>Mobile: 404 Not Found / 422 Unprocessable Entity
        end

        Service->>ClientRepo: existsByTenantIdAndEmail(tenant.id, email)
        ClientRepo->>DB: SELECT count(1) FROM clients WHERE tenant_id = ? AND email = ?
        DB-->>ClientRepo: count
        ClientRepo-->>Service: boolean (true/false)

        alt Email Already Exists in Tenant
            Service-->>Controller: throw EmailAlreadyRegisteredException
            Controller-->>Mobile: 409 Conflict ("Email already registered for this gym")
        end

        Service->>Encoder: encode(rawPassword)
        Encoder-->>Service: passwordHash ($2a$12$...)

        Service->>Service: Client.create(tenant.id, email, passwordHash, ...)
        Service->>ClientRepo: save(client)
        ClientRepo->>DB: INSERT INTO clients (...) VALUES (...)
        DB-->>ClientRepo: Persisted Client

        Service->>Service: Subscription.createInitialPending(tenant.id, client.id)
        Service->>SubRepo: save(subscription)
        SubRepo->>DB: INSERT INTO subscriptions (...) VALUES (...)

  ### 4. Physical Database Schema (PostgreSQL DDL)

    -- 1. Tenants Table
    CREATE TABLE IF NOT EXISTS gym_tenants (
        id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        tenant_code VARCHAR(32) NOT NULL UNIQUE,
        gym_name VARCHAR(128) NOT NULL,
        subdomain VARCHAR(64) UNIQUE,
        slot_booking_enabled BOOLEAN NOT NULL DEFAULT TRUE,
        cancellation_window_hours INT NOT NULL DEFAULT 24,
        monthly_debt_cutoff_day INT NOT NULL DEFAULT 10,
        is_active BOOLEAN NOT NULL DEFAULT TRUE,
        created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP
    );

    -- 2. Clients Table (Scoped by Tenant)
    CREATE TABLE IF NOT EXISTS clients (
        id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        tenant_id UUID NOT NULL REFERENCES gym_tenants(id) ON DELETE RESTRICT,
        email VARCHAR(255) NOT NULL,
        password_hash VARCHAR(255) NOT NULL,
        full_name VARCHAR(128) NOT NULL,
        phone_number VARCHAR(32) NOT NULL,
        emergency_contact VARCHAR(255) NOT NULL,
        enrollment_date DATE NOT NULL DEFAULT CURRENT_DATE,
        account_status VARCHAR(32) NOT NULL DEFAULT 'PENDING_PAYMENT',
        created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
        updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
        CONSTRAINT uq_tenant_client_email UNIQUE (tenant_id, email)
    );

    CREATE INDEX idx_clients_tenant_id ON clients(tenant_id);
    CREATE INDEX idx_clients_email ON clients(email);

    -- 3. Initial Subscriptions Table
    CREATE TABLE IF NOT EXISTS subscriptions (
        id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        tenant_id UUID NOT NULL REFERENCES gym_tenants(id) ON DELETE RESTRICT,
        client_id UUID NOT NULL REFERENCES clients(id) ON DELETE CASCADE,
        membership_plan_id UUID NULL, -- Assigned during UC-02 plan selection
        status VARCHAR(32) NOT NULL DEFAULT 'PENDING',
        payment_type VARCHAR(32) NULL, -- 'RECURRING' | 'MANUAL'
        is_in_debt BOOLEAN NOT NULL DEFAULT FALSE,
        external_mp_preapproval_id VARCHAR(128) NULL,
        start_date DATE NULL,
        current_period_end DATE NULL,
        created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
        updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP
    );

    CREATE INDEX idx_subscriptions_tenant_client ON subscriptions(tenant_id, client_id);
